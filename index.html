Le cerveau de votre bot : 
// Exemple simplifié de logique
client.on('message', async (m) => {
    const isGroup = m.key.remoteJid.endsWith('@g.us');
    const sender = m.key.participant;
    const body = m.message.conversation || m.message.extendedTextMessage?.text;

    if (isGroup && body.includes('http')) {
        // Supprimer le message
        await client.sendMessage(m.key.remoteJid, { delete: m.key });
        
        // Envoyer l'avertissement
        await client.sendMessage(m.key.remoteJid, { 
            text: `⚠️ @${sender.split('@')[0]} ⚠️ les liens ne sont pas autorisés sans l'accord des administrateurs`,
            mentions: [sender]
        });
    }
});
